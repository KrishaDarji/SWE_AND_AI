```python
def normalize_rows(rows):
    """Validate reusable record input without mutating it; ignore no unknown keys."""
    if not isinstance(rows, (list, tuple)) or not rows:
        raise ValueError('rows must be a nonempty list or tuple of record dictionaries')
    clean = []
    for row in rows:
        if not isinstance(row, dict) or set(row) - (set(CONFIGURATION) | set(OBJECTIVES)):
            raise ValueError('Record must contain only Health schema fields')
        if 'MRE-' not in row:
            raise ValueError('Each record requires MRE-; use None for missing')
        config = validate_configuration({name: row.get(name) for name in CONFIGURATION})
        target = validate_configuration({'N_estimators': row['MRE-']})['N_estimators']
        clean.append({**config, 'MRE-': target})
    return clean


def partition(values, categorical=False):
    """Return JSON definitions and group indices; missing predictor index is None."""
    ordered = sorted(v for v in values if v is not None)
    distinct = sorted(set(ordered))
    if categorical or len(distinct) <= 8:
        lookup = {v: i for i, v in enumerate(distinct)}
        return {'kind': 'categories' if categorical else 'exact_values', 'cuts': [],
                'groups': [{'value': v} for v in distinct]}, [lookup.get(v) for v in values]
    cuts = sorted({quantile(ordered, p) for p in (.25, .5, .75)} - {ordered[-1]})
    bounds = [ordered[0], *cuts, ordered[-1]]
    definitions = [{'lower': a, 'upper': b, 'lower_inclusive': i == 0,
                    'upper_inclusive': True} for i, (a, b) in enumerate(zip(bounds, bounds[1:]))]
    return {'kind': 'quartile_ranges', 'cuts': cuts, 'groups': definitions}, [
        None if v is None else bisect_left(cuts, v) for v in values]


def summary(values):
    return observed_summary(values, len(values))


def diagnose(rows):
    """Rank only five configuration fields; report observations, not interventions."""
    rows = normalize_rows(rows)
    fields, memberships = {}, {}
    for name in CONFIGURATION:
        values = [r[name] for r in rows]
        grouping, ids = partition(values, name == 'criterion')
        memberships[name] = ids
        groups = []
        for group_id, definition in enumerate(grouping['groups']):
            outcomes = [r['MRE-'] for r, g in zip(rows, ids) if g == group_id]
            stats = summary(outcomes)
            groups.append({'group_id': group_id, 'definition': definition,
                           'row_count': len(outcomes), 'mre': stats, 'sparse': stats['n'] < 5})
        included = [r['MRE-'] for r, g in zip(rows, ids) if g is not None and r['MRE-'] is not None]
        overall = summary(included)
        labeled = [g for g in groups if g['mre']['n']]
        status = ('all_missing_predictor' if not grouping['groups'] else
                  'constant_predictor' if len(set(v for v in values if v is not None)) == 1 else
                  'insufficient_labeled_groups' if len(labeled) < 2 else 'ok')
        mean = overall['mean']
        ss_total = math.fsum((y - mean) ** 2 for y in included) if included else 0.0
        ss_between = math.fsum(g['mre']['n'] * (g['mre']['mean'] - mean) ** 2 for g in labeled) if labeled else 0.0
        if status == 'ok' and ss_total == 0:
            status = 'constant_outcome'
        score = ss_between / ss_total if status == 'ok' else None
        for g in groups:
            delta = g['mre']['mean'] - mean if g['mre']['n'] else None
            g['mean_minus_included_mean'] = delta
            g['observed_direction'] = ('unavailable' if delta is None else 'equal' if delta == 0
                                       else 'higher_MRE' if delta > 0 else 'lower_MRE')
        low = min((g['mre']['mean'] for g in labeled), default=None)
        high = max((g['mre']['mean'] for g in labeled), default=None)
        fields[name] = {'field': name, 'rank': None, 'status': status, 'eta_squared': score,
                        'ss_between': ss_between, 'ss_total': ss_total,
                        'nonmissing_predictor_count': len(values) - values.count(None),
                        'distinct_nonmissing_values': len(set(v for v in values if v is not None)),
                        'grouping': grouping, 'groups': groups, 'included_mre': overall,
                        'missing_predictor_mre': summary([r['MRE-'] for r in rows if r[name] is None]),
                        'highest_mean_group_ids': [g['group_id'] for g in labeled if g['mre']['mean'] == high],
                        'lowest_mean_group_ids': [g['group_id'] for g in labeled if g['mre']['mean'] == low]}
    ranked = sorted((f for f in fields.values() if f['eta_squared'] is not None),
                    key=lambda f: -f['eta_squared'])
    for f in ranked:
        f['rank'] = 1 + sum(other['eta_squared'] > f['eta_squared'] for other in ranked)
    joint = []
    for a, b in combinations(CONFIGURATION, 2):
        cells = {}
        for r, ga, gb in zip(rows, memberships[a], memberships[b]):
            if ga is not None and gb is not None:
                cells.setdefault((ga, gb), []).append(r['MRE-'])
        joint.append({'fields': [a, b], 'excluded_missing_predictor_rows': len(rows) - sum(map(len, cells.values())),
                      'cells': [{'group_ids': list(key), 'row_count': len(ys),
                                 'mre': summary(ys), 'sparse': sum(y is not None for y in ys) < 5}
                                for key, ys in sorted(cells.items())]})
    report = {'method': METHOD, 'target': {'field': 'MRE-', 'direction': 'minimize'},
              'predictor_fields': list(CONFIGURATION), 'excluded_predictors': list(OBJECTIVES),
              'record_count': len(rows), 'overall_mre': summary([r['MRE-'] for r in rows]),
              'repeated_configurations': duplicates(tuple(r[n] for n in CONFIGURATION) for r in rows),
              'ranked_findings': ranked,
              'unranked_fields': [f for f in fields.values() if f['eta_squared'] is None],
              'pairwise_group_summaries': joint, 'limitations': LIMITATIONS,
              'environment': {'scikit_learn_version': sklearn.__version__},
              'reuse': 'Unchanged Step 1 loader, quantiles and duplicate counts; Step 2 configuration normalization; Step 3 observed summaries. Steps 1–4 source and tests retained unchanged. Added grouped marginal ranking, directions, missing/sparse/constant handling and joint summaries.'}
    report['maintainer_explanation'] = render_text(report)
    json.dumps(report, allow_nan=False)
    return report

def main():
```

```python
"""Compare two predeclared observed configuration families, without model predictions."""
import argparse
import json
import math
import platform
import statistics
from collections import Counter
from pathlib import Path

import scipy
import sklearn
from scipy.stats import t

from summarize import CONFIGURATION, OBJECTIVES, duplicates, load_rows, quantile


def observed_summary(values, total):
    """Summarize finite observed values; missingness is counted per objective."""
    clean = []
    for value in values:
        if value is None:
            continue
        if isinstance(value, bool) or not isinstance(value, (int, float)) or not math.isfinite(value):
            raise ValueError('Outcomes must be finite numbers or None')
        clean.append(float(value))
    ordered = sorted(clean)
    return {'n': len(clean), 'missing': total - len(clean),
            'mean': statistics.mean(clean) if clean else None,
            'std_sample': statistics.stdev(clean) if len(clean) > 1 else None,
            'median': statistics.median(clean) if clean else None,
            'q25': quantile(ordered, .25) if clean else None,
            'q75': quantile(ordered, .75) if clean else None,
            'min': min(clean) if clean else None, 'max': max(clean) if clean else None}


def welch_comparison(baseline, candidate, direction='minimize', practical_threshold=None):
    """Two-sided Welch test of equal population means; CI is candidate minus baseline.

    A threshold, if supplied, is a positive minimum meaningful improvement in the
    recorded objective scale. It must be justified externally before interpreting
    results. This function does not treat a nonsignificant test as equivalence.
    """
    if direction not in ('minimize', 'maximize'):
        raise ValueError('direction must be minimize or maximize')
    if practical_threshold is not None and (isinstance(practical_threshold, bool) or
            not isinstance(practical_threshold, (int, float)) or
            not math.isfinite(practical_threshold) or practical_threshold <= 0):
        raise ValueError('practical_threshold must be positive and finite')
    a = observed_summary(baseline, len(baseline))
    b = observed_summary(candidate, len(candidate))
    delta = b['mean'] - a['mean'] if a['n'] and b['n'] else None
    improvement = None if delta is None else (-delta if direction == 'minimize' else delta)
    result = {'candidate_minus_baseline_mean': delta, 'objective_direction': direction,
              'observed_direction': ('unavailable' if delta is None else 'equal' if delta == 0
                                     else 'candidate_better' if improvement > 0 else 'candidate_worse'),
              'status': 'insufficient_data', 'standard_error': None, 'degrees_of_freedom': None,
              't_statistic': None, 'p_two_sided': None, 'confidence_level': .95,
              'confidence_interval': None, 'alpha': .05, 'statistically_detected': None,
              'practical_threshold': practical_threshold,
              'practical_assessment': 'Undetermined: no externally justified minimum meaningful improvement supplied.'
              if practical_threshold is None else 'Unassessed: inferential interval unavailable.'}
    if min(a['n'], b['n']) < 2:
        return result
    va = a['std_sample'] ** 2 / a['n']
    vb = b['std_sample'] ** 2 / b['n']
    variance = va + vb
    if variance == 0:
        result['status'] = 'zero_variance_inference_unavailable'
        return result
    se = math.sqrt(variance)
    df = variance ** 2 / (va ** 2 / (a['n'] - 1) + vb ** 2 / (b['n'] - 1))
    statistic = delta / se
    p = float(2 * t.sf(abs(statistic), df))
    margin = float(t.ppf(.975, df)) * se
    interval = [delta - margin, delta + margin]
    result.update(status='ok', standard_error=se, degrees_of_freedom=df,
                  t_statistic=statistic, p_two_sided=p, confidence_interval=interval,
                  statistically_detected=p < .05)
    if practical_threshold is not None:
        benefit_interval = [-interval[1], -interval[0]] if direction == 'minimize' else interval
        result['practical_assessment'] = (
            'Interval supports improvement exceeding the supplied threshold.'
            if benefit_interval[0] > practical_threshold else
            'Interval does not support improvement exceeding the supplied threshold; this is not an equivalence test.')
    return result


def select_families(rows, field, baseline, candidate):
    """Membership uses only the declared configuration field, never objectives."""
    if field not in CONFIGURATION or baseline is None or candidate is None or baseline == candidate:
        raise ValueError('Use a configuration field and two distinct nonmissing family values')
    groups = {'baseline': [], 'candidate': []}
    excluded = []
    for i, row in enumerate(rows):
        value = row[field]
        label = 'baseline' if value == baseline else 'candidate' if value == candidate else None
        if label is None:
            excluded.append(i)
        else:
            groups[label].append(i)
    return groups, excluded


def compare_families(rows, field='criterion', baseline='squared_error', candidate='absolute_error',
                     primary='MRE-', practical_threshold=None):
    """Reusable comparison on validated observed rows; no surrogate is consulted."""
    if primary not in OBJECTIVES:
        raise ValueError('Unknown primary objective')
    groups, excluded = select_families(rows, field, baseline, candidate)
    summaries = {}
    for label, indices in groups.items():
        summaries[label] = {'total_rows': len(indices), 'row_ids': [i + 1 for i in indices],
                            'objectives': {name: observed_summary([rows[i][name] for i in indices], len(indices))
                                           for name in OBJECTIVES},
                            'configuration_repetitions': duplicates(tuple(rows[i][name] for name in CONFIGURATION)
                                                                    for i in indices)}
    comparison = welch_comparison([rows[i][primary] for i in groups['baseline']],
                                 [rows[i][primary] for i in groups['candidate']],
                                 OBJECTIVES[primary], practical_threshold)
    differences = {name: (summaries['candidate']['objectives'][name]['mean'] -
                         summaries['baseline']['objectives'][name]['mean'])
                   if all(summaries[label]['objectives'][name]['n'] for label in groups) else None
                   for name in OBJECTIVES}
    if comparison['status'] != 'ok':
        evidence = 'Inference unavailable because of insufficient observations or zero estimated variance.'
    elif comparison['statistically_detected']:
        evidence = 'A mean difference is detected under the stated sampling assumptions.'
    else:
        evidence = 'No mean difference is detected at alpha 0.05; this does not prove equivalence.'
    conclusion = (f"Observed primary direction: {comparison['observed_direction']}. {evidence} "
                  f"{comparison['practical_assessment']} This comparison does not certify a causal "
                  'improvement or repeatability for a particular configuration.')
    return {'group_definitions': {'field': field, 'baseline': baseline, 'candidate': candidate,
                                 'selection': 'Fixed before examining outcomes; membership uses configuration only'},
            'record_count': len(rows), 'included_count': sum(map(len, groups.values())),
            'excluded_count': len(excluded),
            'excluded_family_counts': [{'value': value, 'count': count} for value, count in
                                       Counter(rows[i][field] for i in excluded).items()],
            'excluded_missing_family_count': sum(rows[i][field] is None for i in excluded),
            'groups': summaries, 'primary_objective': primary,
            'candidate_minus_baseline_means': differences, 'primary_comparison': comparison,
            'secondary_objectives': {'names': [name for name in OBJECTIVES if name != primary],
                                     'scope': 'Descriptive only; no secondary tests, intervals, or significance claims'},
            'method': {'name': 'Two-sided Welch unequal-variance t test and 95% Welch t interval for mean difference',
```

```python
    'invalid': 'Invalid arrivals raise ValueError without changing state. Finite inputs causing non-finite moment arithmetic are also rejected.',

    'Missing outcomes may be selective; each objective uses its own observed subset.',
    'Recorded metric scales and directions only; formulas, physical units, compute costs, and theoretical ranges are unspecified.',
    'Float arithmetic can round near ties and running moments; extreme magnitudes overflowing moment arithmetic are rejected.',
    'Configuration validation reuses Step 2 normalization; absent configuration fields are permitted, with no invented hyperparameter domains.',

def normalize_outcomes(outcomes):
    """Reuse finite-number normalization, with objective-specific keys."""
    if not isinstance(outcomes, dict) or set(outcomes) - set(OBJECTIVES):
        raise ValueError('Outcomes must be a dictionary containing only objective keys')
    return {name: validate_configuration({'N_estimators': outcomes.get(name)})['N_estimators']
            for name in OBJECTIVES}


class FlowEvaluator:
    """Consume one arrival per call, using O(number of objectives) state."""

    def __init__(self):
        self._received = 0
        self._state = {name: {'observed_count': 0, 'missing_count': 0,
                             'mean': None, 'm2': 0.0, 'best': None, 'best_tie_count': 0}
                       for name in OBJECTIVES}

    def snapshot(self):
        """Return detached, JSON-ready current summaries, without mutation."""
        objectives = copy.deepcopy(self._state)
        for name, state in objectives.items():
            count = state['observed_count']
            m2 = state.pop('m2')
            variance = max(0.0, m2) / (count - 1) if count > 1 else None
            state.update(direction=OBJECTIVES[name], variance_sample=variance,
                         std_sample=math.sqrt(variance) if variance is not None else None)
        return {'configurations_received': self._received, 'objectives': objectives}

    def update(self, configuration, outcomes):
        """Validate atomically, update Welford moments, and return a snapshot."""
        config = validate_configuration(configuration)
        values = normalize_outcomes(outcomes)
        states = copy.deepcopy(self._state)
        arrival = self._received + 1
        for name, value in values.items():
            state = states[name]
            if value is None:
                state['missing_count'] += 1
                continue
            state['observed_count'] += 1
            count = state['observed_count']
            if count == 1:
                state['mean'] = value
            else:
                delta = value - state['mean']
                state['mean'] += delta / count
                state['m2'] += delta * (value - state['mean'])
                if not all(math.isfinite(state[key]) for key in ('mean', 'm2')):
                    raise ValueError('Arrival exceeds finite running-moment arithmetic')
            best = state['best']
            better = (best is None or
                      (value < best['value'] if OBJECTIVES[name] == 'minimize'
                       else value > best['value']))
            if better:
                state['best'] = {'arrival_index': arrival, 'value': value,
                                 'configuration': config.copy()}
                state['best_tie_count'] = 1
            elif value == best['value']:
                state['best_tie_count'] += 1
        self._state = states
        self._received = arrival
        return self.snapshot()


def iter_arrivals(dataset):
    """Validate the reused schema and reveal one CSV record at a time."""
    with Path(dataset).open(encoding='utf-8-sig', newline='') as handle:
        reader = csv.reader(handle)
        headers = next(reader, None)
        if not headers or len(headers) != len(set(headers)) or set(headers) != EXPECTED:
            raise ValueError('Expected exactly the eight unique Health column names')
        received = False
        for line, cells in enumerate(reader, 2):
            if len(cells) != len(headers):
                raise ValueError(f'Row {line}: expected {len(headers)} cells')
            row = dict(zip(headers, cells))
            try:
                config = validate_configuration({name: row[name] for name in CONFIGURATION})
                outcomes = normalize_outcomes({name: row[name] for name in OBJECTIVES})
            except ValueError as exc:
                raise ValueError(f'Row {line}: {exc}') from exc
            received = True
            yield config, outcomes
        if not received:
            raise ValueError('Dataset must contain at least one record')


def replay(dataset):
    """Retain per-arrival snapshots; never consult later records in an update."""
    evaluator = FlowEvaluator()
    history = []
    for config, outcomes in iter_arrivals(dataset):
        snapshot = evaluator.update(config, outcomes)
        history.append({'arrival_index': snapshot['configurations_received'],
                        'configuration': config, 'observed_objectives': outcomes,
                        'summary': snapshot})
    return {'dataset_path': str(Path(dataset).resolve()),
            'configuration_fields': list(CONFIGURATION),
            'objective_directions': dict(OBJECTIVES), 'arrival_rules': dict(ARRIVAL_RULES),
            'history': history, 'final_summary': evaluator.snapshot(),
            'limitations': list(LIMITATIONS),
            'environment': {'scikit_learn_version': sklearn.__version__}}
```

```python
SETTINGS = dict(n_estimators=120, max_depth=12, min_samples_leaf=2,

def validate_configuration(value):
    """Require only known predictor keys; omitted/null/blank fields mean missing."""
    if not isinstance(value, dict):
        raise ValueError('Configuration must be a JSON object')
    extra = set(value) - set(CONFIGURATION)
    if extra:
        raise ValueError(f'Objective or unknown configuration fields: {sorted(extra)}')
    result = {}
    for name in CONFIGURATION:
        cell = value.get(name)
        if cell is None or (isinstance(cell, str) and not cell.strip()):
            result[name] = None
        elif name == 'criterion':
            if not isinstance(cell, str):
                raise ValueError('criterion must be a string or null')
            result[name] = cell.strip()
        else:
            if isinstance(cell, bool) or not isinstance(cell, (str, int, float)):
                raise ValueError(f'{name} must be a finite number or null')
            try:
                number = float(cell)
            except (ValueError, OverflowError) as exc:
                raise ValueError(f'{name} must be a finite number') from exc
            if not math.isfinite(number):
                raise ValueError(f'{name} must be finite')
            result[name] = number
    return result


def feature_matrix(rows):
    """Select only the five predictors, regardless of other keys in CSV rows."""
    return np.asarray([[row[name] if row[name] is not None else
                        (None if name == 'criterion' else np.nan)
                        for name in CONFIGURATION] for row in rows], dtype=object)


def make_model():
    numeric = SimpleImputer(strategy='median', keep_empty_features=True)
    category = Pipeline([
        ('impute', SimpleImputer(missing_values=None, strategy='most_frequent',
                                 keep_empty_features=True)),
        ('encode', OneHotEncoder(handle_unknown='ignore', sparse_output=False))])
    preprocess = ColumnTransformer([
        ('numeric', numeric, NUMERIC), ('category', category, [1])], remainder='drop')
    return Pipeline([('preprocess', preprocess),
                     ('regressor', RandomForestRegressor(**SETTINGS))])


def split_rows(rows):
    """Group on normalized predictors alone, holding out ceil(20% of groups)."""
    mapping = {}
    groups = []
    for row in rows:
        key = tuple(row[name] for name in CONFIGURATION)
        if key not in mapping:
            mapping[key] = len(mapping)
        groups.append(mapping[key])
    if len(mapping) < 2:
        raise ValueError('At least two distinct normalized configurations required')
    train, heldout = next(GroupShuffleSplit(n_splits=1, test_size=.2,
                                            random_state=42).split(rows, groups=groups))
    return train, heldout, groups


def fit_evaluate(rows):
    """Fit per-target pipelines on labeled training rows; evaluate only afterward."""
    train, heldout, groups = split_rows(rows)
    features = feature_matrix(rows)
    models, results = {}, {}
    for target in TARGETS:
        labeled_train = [int(i) for i in train if rows[i][target] is not None]
        labeled_test = [int(i) for i in heldout if rows[i][target] is not None]
        if not labeled_train or not labeled_test:
            raise ValueError(f'{target}: labeled training and held-out rows required')
        y_train = np.asarray([rows[i][target] for i in labeled_train])
        model = make_model().fit(features[labeled_train], y_train)
        baseline = float(np.mean(y_train))
        predicted = model.predict(features[labeled_test])
        actual = [rows[i][target] for i in labeled_test]
        pre = model.named_steps['preprocess']
        results[target] = {
            'direction': TARGETS[target], 'training_sample_size': len(labeled_train),
            'heldout_sample_size': len(labeled_test),
            'training_row_ids': [i + 1 for i in labeled_train],
            'heldout_row_ids': [i + 1 for i in labeled_test],
            'training_mean_baseline': baseline,
            'training_target_range': [float(min(y_train)), float(max(y_train))],
            'model_mae': float(mean_absolute_error(actual, predicted)),
            'baseline_mae': float(mean_absolute_error(actual, np.full(len(actual), baseline))),
            'learned_numeric_imputation': pre.named_transformers_['numeric'].statistics_.tolist(),
            'learned_criterion_categories': pre.named_transformers_['category'].named_steps['encode'].categories_[0].tolist(),
            'heldout_predictions': [dict(row_id=i + 1, actual=y, predicted=float(p), baseline=baseline)
                                    for i, y, p in zip(labeled_test, actual, predicted)]}
        models[target] = model
    identity = lambda indices: [dict(row_id=int(i) + 1, csv_line=int(i) + 2,
                                     group_id=groups[i]) for i in indices]
    split = {'seed': 42, 'test_group_fraction': .2,
             'method': 'GroupShuffleSplit on first-seen IDs of normalized predictor tuples',
             'row_count': len(rows), 'group_count': len(set(groups)),
             'training_row_count': len(train), 'heldout_row_count': len(heldout),
             'training_group_count': len({groups[i] for i in train}),
             'heldout_group_count': len({groups[i] for i in heldout}),
             'training': identity(train), 'heldout': identity(heldout),
             'groups': [dict(group_id=g, configuration={n: r[n] for n in CONFIGURATION})
                        for g, r in sorted({groups[i]: r for i, r in enumerate(rows)}.items())]}
    return models, {'target_definitions': {t: {'direction': d, 'definition': 'Recorded CSV column; formula unspecified'} for t, d in TARGETS.items()},
                    'predictor_fields': list(CONFIGURATION), 'excluded_objectives': list(OBJECTIVES),
                    'model_settings': {'estimator': 'Independent RandomForestRegressor per target', **SETTINGS},
                    'preprocessing': {'numeric_fields': [CONFIGURATION[i] for i in NUMERIC],
                                      'numeric': 'Training median; keep empty columns with zero fallback',
                                      'criterion': 'Training mode imputation then one-hot; unknown categories ignored',
                                      'fit_scope': 'Labeled training rows separately for each target'},
                    'versions': {'python': platform.python_version(), 'scikit_learn': sklearn.__version__,
                                 'numpy': np.__version__, 'scipy': scipy.__version__},
                    'split': split, 'targets': results, 'limitations': LIMITATIONS}


def predict_configuration(models, configuration):
    clean = validate_configuration(configuration)
    features = feature_matrix([clean])
    return {target: float(models[target].predict(features)[0]) for target in TARGETS}


def unique_json_object(pairs):
    result = {}
    for name, value in pairs:

            raise ValueError(f'Duplicate JSON field: {name}')
        result[name] = value
    return result

                      'heldout_mae': {t: {'model': r['model_mae'], 'training_mean_baseline': r['baseline_mae']}
                                      for t, r in report['targets'].items()}}, indent=2, allow_nan=False))
```

```python
def normalize_records(rows):
    """Normalize reusable inputs; omitted schema fields mean missing."""
    if not isinstance(rows, (list, tuple)) or not rows:
        raise ValueError('rows must be a nonempty list or tuple')
    normalized = []

            raise ValueError('Records must be dictionaries containing only Health fields')

def selection_key(item):
    row_id, row = item
    return row['MRE-'], -row['ACC+'], -row['PRED40+'], row_id


def difference(value, reference, direction):
    """Keep raw difference and direction-aware improvement separate."""
    delta = value - reference
    benefit = -delta if direction == 'minimize' else delta
    return {'selected_minus_reference': delta, 'improvement_in_objective_direction': benefit,
            'observed_relation': 'equal' if benefit == 0 else 'better' if benefit > 0 else 'worse'}
```

```python
"""Describe Health CSV configurations and objectives without fitting a model."""
import argparse
import csv
import json
import math
import statistics
from collections import Counter
from pathlib import Path

import sklearn

CONFIGURATION = ('N_estimators', 'criterion', 'Min_sample_leaves',
                 'Min_impurity_decrease', 'Max_depth')
OBJECTIVES = {'MRE-': 'minimize', 'ACC+': 'maximize', 'PRED40+': 'maximize'}
CATEGORICAL = {'criterion'}
EXPECTED = set(CONFIGURATION) | set(OBJECTIVES)


def quantile(values, fraction):
    """Compute a linearly interpolated quantile on sorted, nonempty values."""
    position = (len(values) - 1) * fraction
    lower = math.floor(position)
    upper = math.ceil(position)
    return values[lower] + (values[upper] - values[lower]) * (position - lower)


def duplicates(keys):
    """Count groups, participating records, and extra records separately."""
    counts = Counter(keys)
    return {'unique_count': len(counts),
            'repeated_group_count': sum(n > 1 for n in counts.values()),
            'records_in_repeated_groups': sum(n for n in counts.values() if n > 1),
            'extra_record_count': sum(n - 1 for n in counts.values())}


def load_rows(dataset):
    """Validate and normalize the shared eight-column Health CSV schema."""
    rows = []
    with Path(dataset).open(encoding='utf-8-sig', newline='') as handle:
        reader = csv.reader(handle)
        headers = next(reader, None)
        if not headers or len(headers) != len(set(headers)) or set(headers) != EXPECTED:
            raise ValueError('Expected exactly the eight unique Health column names')
        for line, raw in enumerate(reader, 2):
            if len(raw) != len(headers):
                raise ValueError(f'Row {line}: expected {len(headers)} cells')
            row = {}
            for name, cell in zip(headers, raw):
                cell = cell.strip()
                if not cell:
                    row[name] = None
                elif name in CATEGORICAL:
                    row[name] = cell
                else:
                    try:
                        number = float(cell)
                    except ValueError as exc:
                        raise ValueError(f'Row {line}, {name}: invalid numeric value') from exc
                    if not math.isfinite(number):
                        raise ValueError(f'Row {line}, {name}: non-finite numeric value')
                    row[name] = number
            rows.append(row)
    if not rows:
        raise ValueError('Dataset must contain at least one record')
    return headers, rows


def summarize(dataset):
    """Validate the schema and return a JSON-serializable descriptive report."""
    headers, rows = load_rows(dataset)
    columns = {}
    for name in headers:
        values = [row[name] for row in rows if row[name] is not None]
        distinct = sorted(set(values))
        info = {'role': 'objective' if name in OBJECTIVES else 'configuration',
                'type': 'categorical' if name in CATEGORICAL else 'numeric',
                'nonmissing_count': len(values), 'missing_count': len(rows) - len(values),
                'distinct_count': len(distinct), 'distinct_values': distinct}
        if name in CATEGORICAL:
            info['value_counts'] = dict(sorted(Counter(values).items()))
        else:
            ordered = sorted(values)
            info['statistics'] = None if not values else {
                'count': len(values), 'mean': statistics.mean(values),
                'std_sample': statistics.stdev(values) if len(values) > 1 else None,
                'min': ordered[0], 'q25': quantile(ordered, .25),
                'median': statistics.median(ordered), 'q75': quantile(ordered, .75),
                'max': ordered[-1]}
        if name in OBJECTIVES:
            info['direction'] = OBJECTIVES[name]
            best = (min(values) if OBJECTIVES[name] == 'minimize' else max(values)) if values else None
            info['best_observed'] = best
            info['best_observed_count'] = values.count(best) if values else 0
        columns[name] = info
    return {'dataset_path': str(Path(dataset).resolve()),
            'record_count': len(rows), 'column_count': len(headers), 'column_names': headers,
            'configuration_fields': list(CONFIGURATION), 'objective_directions': OBJECTIVES,
            'columns': columns,
            'duplicate_records': duplicates(tuple(row[n] for n in headers) for row in rows),
            'repeated_configurations': duplicates(tuple(row[n] for n in CONFIGURATION) for row in rows),
            'environment': {'scikit_learn_version': sklearn.__version__},
            'conventions': {
                'missing': 'Empty or whitespace-only cells; other numeric tokens must be finite numbers.',
                'distinct': 'Nonmissing normalized values only; numeric values parsed as float.',
                'quantiles': 'Linear interpolation at (n - 1) * p.',
                'std': 'Sample standard deviation (n - 1); null for fewer than two values.',
                'duplicates': 'After trimming whitespace and parsing numbers; missing cells compare equal.',
                'scope': 'Descriptive observed metrics only; no training, formulas, units, or timeline inferred.'}}


def render_text(report):
    lines = [f"Records: {report['record_count']}; columns: {report['column_count']}"]
    for role in ('configuration', 'objective'):
        lines.append(f'\n{role.title()} fields:')
        for name, info in report['columns'].items():
            if info['role'] != role:
                continue
            lines.append(f"{name}: {info['type']}; missing={info['missing_count']}; distinct={info['distinct_count']}")
            if info['type'] == 'categorical':
                lines.append('  Value counts: ' + json.dumps(info['value_counts'], sort_keys=True))
            else:
                lines.append('  Statistics: ' + json.dumps(info['statistics'], sort_keys=True))
                if role == 'configuration':
                    lines.append('  Distinct values: ' + json.dumps(info['distinct_values']))
            if role == 'objective':
                lines.append(f"  Direction: {info['direction']}; best observed={info['best_observed']}; count={info['best_observed_count']}")
    for key in ('duplicate_records', 'repeated_configurations'):
        lines.append(f'\n{key}: ' + json.dumps(report[key], sort_keys=True))
    lines.append('\nFull distinct values for every column are in summary.json.')
    lines.append('\nConventions: ' + json.dumps(report['conventions'], sort_keys=True))
    return '\n'.join(lines) + '\n'


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('dataset', type=Path)
    parser.add_argument('--output-dir', required=True, type=Path)
    args = parser.parse_args()
    try:
        report = summarize(args.dataset)
    except (OSError, ValueError, csv.Error) as exc:
        parser.exit(2, f'Error: {exc}\n')
    args.output_dir.mkdir(parents=True, exist_ok=True)
    (args.output_dir / 'summary.json').write_text(json.dumps(report, indent=2, allow_nan=False) + '\n', encoding='utf-8')
    rendered = render_text(report)
    (args.output_dir / 'summary.txt').write_text(rendered, encoding='utf-8')
    print(rendered, end='')


if __name__ == '__main__':
    main()
```
