```python
"""Predict Performance- using encoder settings and a reproducible holdout."""

import argparse
import json
import math
from pathlib import Path
import platform

import sklearn
from sklearn.dummy import DummyRegressor
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error
from sklearn.model_selection import GroupShuffleSplit

from remember import OBJECTIVE, load_measurements


SEED = 42
TEST_SIZE = 0.2
MODEL_SETTINGS = dict(n_estimators=200, criterion="squared_error", max_depth=None,
                      min_samples_split=2, min_samples_leaf=1, max_features=1.0,
                      bootstrap=True, random_state=SEED, n_jobs=1)


def evaluate(data):
    """Fit once on training rows; repeated configurations stay in one partition."""
    features = [list(row[:-1]) for row in data.rows]
    targets = [row[-1] for row in data.rows]
    identities = {key: index for index, key in enumerate(sorted(set(map(tuple, features))))}
    groups = [identities[tuple(row)] for row in features]
    if len(identities) < 2:
        raise ValueError("Evaluation needs at least two distinct configurations")
    splitter = GroupShuffleSplit(n_splits=1, test_size=TEST_SIZE, random_state=SEED)
    train, test = next(splitter.split(features, groups=groups))
    x_train = [features[i] for i in train]
    y_train = [targets[i] for i in train]
    x_test = [features[i] for i in test]
    y_test = [targets[i] for i in test]
    model = RandomForestRegressor(**MODEL_SETTINGS).fit(x_train, y_train)
    baseline = DummyRegressor(strategy="mean").fit(x_train, y_train)
    predicted = model.predict(x_test)
    baseline_predicted = baseline.predict(x_test)
    report = {
        "objective": OBJECTIVE, "direction": "minimize", "units": None,
        "features": list(data.settings),
        "versions": {"python": platform.python_version(), "scikit_learn": sklearn.__version__},
        "split": {"method": "GroupShuffleSplit", "seed": SEED,
                  "test_fraction_of_unique_configurations": TEST_SIZE,
                  "train_count": len(train), "test_count": len(test),
                  "train_data_rows": [int(i) + 1 for i in train],
                  "test_data_rows": [int(i) + 1 for i in test]},
        "model": {"class": "RandomForestRegressor", "settings": dict(MODEL_SETTINGS)},
        "preprocessing": "None; finite numeric settings in CSV column order",
        "model_mae": float(mean_absolute_error(y_test, predicted)),
        "baseline": {"class": "DummyRegressor", "strategy": "mean",
                     "training_mean": float(baseline.constant_[0, 0])},
        "baseline_mae": float(mean_absolute_error(y_test, baseline_predicted)),
        "held_out_predictions": [
            {"data_row": int(i) + 1, "actual": actual, "predicted": float(prediction)}
            for i, actual, prediction in zip(test, y_test, predicted)
        ],
        "limitations": [
            "One fixed holdout; no hyperparameter search or uncertainty calibration.",
            "Random configuration holdout does not measure transfer to new videos or hardware.",
            "Units and flag semantics are unspecified; no causal interpretation.",
            "Constant settings have no measured effect; extrapolation is unsupported.",
            "Predictions retain the training-only fit; evaluation rows are never used to fit.",
            "Per-setting domain checks do not establish validity of combined encoder settings.",
        ],
    }
    return model, report


def predict_configuration(model, data, report, configuration):
    """Require every setting, no target, and values observed in training only."""
    if not isinstance(configuration, dict) or set(configuration) != set(data.settings):
        raise ValueError("Configuration must contain exactly the encoder settings; exclude Performance-")
    train_rows = [data.rows[i - 1] for i in report["split"]["train_data_rows"]]
    values = []
    for index, name in enumerate(data.settings):
        value = configuration[name]
        if isinstance(value, bool) or not isinstance(value, (int, float)) or not math.isfinite(value):
            raise ValueError(f"Setting {name} must be a finite number")
        if value not in {row[index] for row in train_rows}:
            raise ValueError(f"Setting {name} is outside the observed training domain")
        values.append(float(value))
    return {"settings": dict(zip(data.settings, values)),
            "predicted_Performance-": float(model.predict([values])[0]),
            "units": None, "fit": "training partition only"}


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("dataset", type=Path)
    parser.add_argument("--output-dir", required=True, type=Path)
    parser.add_argument("--config", type=Path, help="JSON object with encoder settings only")
    args = parser.parse_args(argv)
    try:
        data = load_measurements(args.dataset)
        model, report = evaluate(data)
        if args.config:
            config = json.loads(args.config.read_text(encoding="utf-8"))
            report["configuration_prediction"] = predict_configuration(model, data, report, config)
        target = args.output_dir / "evaluation.json"
        protected = [args.dataset.resolve()]
        if args.config:
            protected.append(args.config.resolve())
        if target.resolve() in protected:
            raise ValueError("Output must not overwrite an input")
        args.output_dir.mkdir(parents=True, exist_ok=True)
        target.write_text(json.dumps(report, indent=2, allow_nan=False) + "\n", encoding="utf-8")
    except (OSError, ValueError) as error:
        parser.exit(2, f"error: {error}\n")
    print(json.dumps({key: report[key] for key in ("model_mae", "baseline_mae", "split")},
                     sort_keys=True))
    if "configuration_prediction" in report:
        print(json.dumps(report["configuration_prediction"], sort_keys=True))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""Validated descriptive summaries of historical encoder measurements."""

import argparse
import csv
import json
import math
import statistics
from dataclasses import dataclass
from pathlib import Path


OBJECTIVE = "Performance-"


@dataclass(frozen=True)
class Measurements:
    """Numeric records in CSV order; row numbers refer to data rows, starting at 1."""

    settings: tuple[str, ...]
    rows: tuple[tuple[float, ...], ...]


def load_measurements(path):
    """Read numeric settings and the registered objective without changing the CSV."""
    with Path(path).open(newline="", encoding="utf-8-sig") as stream:
        reader = csv.reader(stream)
        header = next(reader, None)
        if not header or any(not name.strip() for name in header):
            raise ValueError("CSV must have nonempty column names")
        if len(set(header)) != len(header):
            raise ValueError("Duplicate column names")
        if OBJECTIVE not in header:
            raise ValueError(f"Missing registered objective: {OBJECTIVE}")
        settings = tuple(name for name in header if name != OBJECTIVE)
        if not settings:
            raise ValueError("CSV must contain encoder settings")
        order = [header.index(name) for name in (*settings, OBJECTIVE)]
        records = []
        for line, cells in enumerate(reader, start=2):
            if len(cells) != len(header):
                raise ValueError(f"CSV line {line}: expected {len(header)} fields")
            try:
                values = tuple(float(cells[index]) for index in order)
            except ValueError as error:
                raise ValueError(f"CSV line {line}: missing or nonnumeric value") from error
            if not all(math.isfinite(value) for value in values):
                raise ValueError(f"CSV line {line}: nonfinite value")
            records.append(values)
    if not records:
        raise ValueError("CSV has no measurement rows")
    return Measurements(settings, tuple(records))


def percentile(values, fraction):
    """Linearly interpolate the sorted sample at (n - 1) * fraction."""
    if not values or not 0 <= fraction <= 1:
        raise ValueError("Percentile needs values and a fraction between 0 and 1")
    ordered = sorted(values)
    position = (len(ordered) - 1) * fraction
    lower = math.floor(position)
    upper = math.ceil(position)
    return ordered[lower] + (ordered[upper] - ordered[lower]) * (position - lower)


def describe(values):
    """Use population standard deviation for the complete supplied history."""
    return {
        "count": len(values),
        "min": min(values),
        "q25": percentile(values, 0.25),
        "median": statistics.median(values),
        "q75": percentile(values, 0.75),
        "max": max(values),
        "mean": statistics.mean(values),
        "population_stddev": statistics.pstdev(values),
    }


def summarize(data, top_n=5):
    """Return JSON-compatible facts; conditional statistics do not imply causation."""
    if top_n < 1:
        raise ValueError("top_n must be positive")
    objectives = [row[-1] for row in data.rows]
    setting_summary = {}
    for index, name in enumerate(data.settings):
        groups = {}
        for row in data.rows:
            groups.setdefault(row[index], []).append(row[-1])
        setting_summary[name] = {
            "constant": len(groups) == 1,
            "binary": set(groups).issubset({0.0, 1.0}),
            "values": [
                {
                    "value": value,
                    "count": len(group),
                    "fraction": len(group) / len(data.rows),
                    "objective": describe(group),
                }
                for value, group in sorted(groups.items())
            ],
        }
    ranked = sorted(enumerate(data.rows, start=1), key=lambda item: item[1][-1])

    def record(item):
        number, row = item
        return {
            "data_row": number,
            "settings": dict(zip(data.settings, row[:-1])),
            OBJECTIVE: row[-1],
        }

    configurations = {row[:-1] for row in data.rows}
    best_value = min(objectives)
    return {
        "row_count": len(data.rows),
        "setting_count": len(data.settings),
        "columns": [*data.settings, OBJECTIVE],
        "objective": {"name": OBJECTIVE, "direction": "minimize", "units": None,
                      "statistics": describe(objectives)},
        "quality_measures": [],
        "quality_note": "No separate quality measures are present in this dataset.",
        "unique_configurations": len(configurations),
        "repeated_configuration_rows": len(data.rows) - len(configurations),
        "duplicate_measurement_rows": len(data.rows) - len(set(data.rows)),
        "settings": setting_summary,
        "best_runs": [record(item) for item in ranked if item[1][-1] == best_value],
        "top_runs": [record(item) for item in ranked[:top_n]],
        "interpretation": [
            "All non-objective columns are treated as recorded encoder settings.",
            "Names and encoded values are preserved; flag semantics are not inferred.",
            "Conditional objective statistics describe associations, not causal effects.",
            "Repeated rows remain in all statistics; no measurements are imputed.",
            "Top runs are ordered by ascending objective, then original CSV order.",
        ],
    }


def render_text(summary):
    stats = summary["objective"]["statistics"]
    lines = [
        f"Historical encoder runs: {summary['row_count']}",
        f"Settings: {summary['setting_count']}",
        f"Unique configurations: {summary['unique_configurations']}",
        f"Repeated configuration rows: {summary['repeated_configuration_rows']}",
        f"Duplicate measurement rows: {summary['duplicate_measurement_rows']}",
        "Objective: Performance- (minimize; units unspecified)",
        "Objective statistics: " + json.dumps(stats, sort_keys=True),
        summary["quality_note"],
        "Setting distributions (value: count; mean Performance-):",
    ]
    for name, setting in summary["settings"].items():
        values = "; ".join(
            f"{item['value']:g}: {item['count']}; mean={item['objective']['mean']:.6f}"
            for item in setting["values"]
        )
        lines.append(f"  {name}{' [constant]' if setting['constant'] else ''}: {values}")
    lines.append(f"Best objective: {stats['min']}; tied runs: {len(summary['best_runs'])}")
    for run in summary["top_runs"]:
        lines.append("  " + json.dumps(run, sort_keys=True))
    lines.extend(summary["interpretation"])
    return "\n".join(lines) + "\n"


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("dataset", type=Path, help="Path to X264_AllMeasurements.csv")
    parser.add_argument("--output-dir", required=True, type=Path)
    parser.add_argument("--top", type=int, default=5)
    args = parser.parse_args(argv)
    try:
        summary = summarize(load_measurements(args.dataset), args.top)
        summary["dataset"] = str(args.dataset)
        report = render_text(summary)
        targets = [args.output_dir / name for name in ("summary.json", "summary.txt")]
        if any(target.resolve() == args.dataset.resolve() for target in targets):
            raise ValueError("Output must not overwrite the dataset")
        args.output_dir.mkdir(parents=True, exist_ok=True)
        targets[0].write_text(json.dumps(summary, indent=2, allow_nan=False) + "\n",
                              encoding="utf-8")
        targets[1].write_text(report, encoding="utf-8")
    except (OSError, ValueError) as error:
        parser.exit(2, f"error: {error}\n")
    print(report, end="")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
        output = self.directory / "reports"
        process = subprocess.run([sys.executable, str(SOURCE), str(path),
                                  "--output-dir", str(output), "--top", "1"],
                                 capture_output=True, text=True)
        self.assertEqual(process.returncode, 0, process.stderr)
        report = json.loads((output / "summary.json").read_text())
        self.assertEqual(report["top_runs"][0]["Performance-"], 2)
        self.assertEqual(len(report["top_runs"]), 1)
        self.assertEqual((output / "summary.txt").read_text(), process.stdout)

    def test_cli_failure_is_actionable(self):
        path = self.dataset("flag,Performance-\n1,nan\n")

                                  "--output-dir", str(self.directory / "reports")],

        self.assertEqual(process.returncode, 2)
        self.assertIn("nonfinite", process.stderr)
        self.assertFalse((self.directory / "reports").exists())

    def test_cli_refuses_to_overwrite_dataset(self):

        path.write_text("flag,Performance-\n1,2\n")
        before = path.read_bytes()
        with self.assertRaises(SystemExit) as result:
            remember.main([str(path), "--output-dir", str(self.directory)])
        self.assertEqual(result.exception.code, 2)
        self.assertEqual(path.read_bytes(), before)

    def test_actual_dataset_matches_independent_csv_calculations(self):
        path = Path(os.environ["X264_DATASET"])
        with path.open(newline="") as stream:
            reader = csv.DictReader(stream)
            columns = reader.fieldnames
            rows = list(reader)
        result = remember.summarize(remember.load_measurements(path))
        objectives = [float(row["Performance-"]) for row in rows]
        self.assertEqual(result["row_count"], 1152)
        self.assertEqual(result["setting_count"], 16)
        self.assertEqual(set(result["columns"]), set(columns))
        self.assertEqual(result["objective"]["statistics"]["min"], min(objectives))
        self.assertAlmostEqual(result["objective"]["statistics"]["mean"],
                               sum(objectives) / len(objectives))
        for name, setting in result["settings"].items():
            self.assertTrue(setting["binary"])
            self.assertEqual(sum(group["count"] for group in setting["values"]), len(rows))
            for group in setting["values"]:
                expected = [float(row["Performance-"]) for row in rows
                            if float(row[name]) == group["value"]]
                self.assertEqual(group["count"], len(expected))
                self.assertAlmostEqual(group["objective"]["mean"], sum(expected) / len(expected))
```
